import math
import pandas as pd
def calculate_pf(num: int, p_a: float) -> float:
    final_res = 0.0
    start_index = (num - 1) // 2 + 1
    for i in range(start_index, num+1):
        final_res += math.comb(num, i) * math.pow(p_a, i) * math.pow(1 - p_a, 
num - i)
    return final_res
def calculate_pall(num: int, p_a: float, num_pre: int) -> float:
    committee_num = num_pre // num
    p_f = calculate_pf(num, p_a)
    return 1 - math.pow(1 - p_f, committee_num)
